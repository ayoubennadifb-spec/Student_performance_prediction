# Define the label encoders for the final categorical features
label_encoders = {}
encoding_options = {
    'Parental_Involvement_encoded': ['Low', 'Medium', 'High'],
    'Access_to_Resources_encoded': ['Low', 'Medium', 'High'],
    'Extracurricular_Activities_encoded': ['Yes', 'No'],
    'Motivation_Level_encoded': ['Low', 'Medium', 'High'],
    'Internet_Access_encoded': ['Yes', 'No'],
    'Teacher_Quality_encoded': ['Low', 'Medium', 'High'],
    'School_Type_encoded': ['Public', 'Private'],
    'Peer_Influence_encoded': ['Positive', 'Neutral', 'Negative'],
    'Parental_Education_Level_encoded': ['High School', 'College', 'Postgraduate'],
    'Distance_from_Home_encoded': ['Near', 'Moderate', 'Far']
}

# Fit the encoders
for feature, options in encoding_options.items():
    encoder = LabelEncoder()
    encoder.fit(options)
    label_encoders[feature] = encoder

# Add sliders for numerical fields and dropdowns for categorical fields
numerical_ranges = {
    "Hours_Studied": (0, 40),
    "Attendance": (0, 100),
    "Sleep_Hours": (0, 24),
    "Previous_Scores": (0, 100),
    "Tutoring_Sessions": (0, 10),
    "Physical_Activity": (0, 10)
}

for label in labels:
    ttk.Label(scrollable_frame, text=label.replace("_encoded", ""), style="TLabel").pack(pady=5)
    if label in numerical_ranges:
        # Create sliders for numerical features
        frame = tk.Frame(scrollable_frame, bg="#ffffff")
        frame.pack(pady=5)

        var = tk.DoubleVar()
        slider = ttk.Scale(frame, from_=numerical_ranges[label][0], to=numerical_ranges[label][1],
                        orient="horizontal", length=300, variable=var)
        slider.pack(side="left", padx=(0, 10))

        # Label pour afficher la valeur arrondie
        int_label_var = tk.StringVar()
        int_label_var.set(str(int(var.get())))

        value_label = tk.Label(frame, textvariable=int_label_var, font=("Calibri", 14), bg="#ffffff", fg="#007BFF")
        value_label.pack(side="left")

        # Fonction de mise à jour lors du mouvement du slider
        def update_label(val, v=var, lv=int_label_var):
            lv.set(str(int(float(val))))

        slider.config(command=update_label)

        entries[label] = var


    elif label in encoding_options:
        # Create dropdowns for categorical features
        var = tk.StringVar(scrollable_frame)
        var.set(encoding_options[label][0])  # Default value
        dropdown = ttk.Combobox(scrollable_frame, textvariable=var, values=encoding_options[label], state="readonly", width=35)
        dropdown.pack(pady=5)
        entries[label] = var

# Prediction function
def predict_score():
    try:
        # Prepare the input data
        data = {}
        for feature in labels:
            value = entries[feature].get()
            if feature in encoding_options:
                # Encode categorical features
                data[feature] = label_encoders[feature].transform([value])[0]
            else:
                # Convert numerical features to float
                data[feature] = float(value)
        
        # Create a DataFrame
        df = pd.DataFrame([data])
        
        # Scale numerical features
        numerical_features = ['Hours_Studied', 'Attendance', 'Sleep_Hours', 'Previous_Scores', 'Tutoring_Sessions', 'Physical_Activity']
        df[numerical_features] = StandardScaler().fit_transform(df[numerical_features])
        
        # Make prediction
        prediction = model.predict(df)[0]
        messagebox.showinfo("🎯 Predicted Score", f"Predicted Exam Score: {prediction:.2f}")
    except Exception as e:
        messagebox.showerror("Error", f"An error occurred: {e}")

# Predict button
predict_button = ttk.Button(scrollable_frame, text="🎯 Predict Score", style="TButton", command=predict_score)

def load_and_predict_batch():
    file_path = filedialog.askopenfilename(filetypes=[("Fichiers Excel/CSV", "*.xlsx *.xls *.csv")])
    if not file_path:
        return

    try:
        # Lire le fichier
        if file_path.endswith(".csv"):
            df = pd.read_csv(file_path)
        else:
            df = pd.read_excel(file_path)

        # Vérifier colonnes nécessaires
        required_columns = list(numerical_ranges.keys()) + list(encoding_options.keys())
        missing_columns = [col for col in required_columns if col not in df.columns]
        if missing_columns:
            messagebox.showerror("Erreur", f"Colonnes manquantes : {', '.join(missing_columns)}")
            return

        # Supprimer les lignes avec données manquantes
        df_clean = df.dropna(subset=required_columns)

        # Encoder les colonnes catégorielles
        for cat_col in encoding_options.keys():
            if cat_col in df_clean.columns:
                df_clean[cat_col] = df_clean[cat_col].map(lambda x: label_encoders[cat_col].transform([x])[0] if x in label_encoders[cat_col].classes_ else None)

        df_clean = df_clean.dropna()  # Supprimer si l'encodage a échoué

        # Normaliser les colonnes numériques
        scaler = StandardScaler()
        df_clean[list(numerical_ranges.keys())] = scaler.fit_transform(df_clean[list(numerical_ranges.keys())])


        # Prédictions
        predictions = model.predict(df_clean)

        # Nettoyer tableau avant ajout
        for row in tree.get_children():
            tree.delete(row)

        # Ajouter résultats
        for i, score in enumerate(predictions):
            tree.insert("", "end", values=(i + 1, f"{score:.2f}"))

        messagebox.showinfo("Succès", f"{len(predictions)} scores prédits avec succès.")
    except Exception as e:
        messagebox.showerror("Erreur", f"Erreur pendant l'import ou la prédiction :\n{str(e)}")

# === BOUTON POUR IMPORTATION DU FICHIER CSV/XLSX ===
